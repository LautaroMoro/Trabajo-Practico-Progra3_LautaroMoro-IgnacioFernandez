```
//  Ejemplos de requests (HTTP)

// Crear producto
POST http://localhost:3000/products
Content-Type: application/json

{
  "descripcion": "Remera básica negra",
  "precio": 9999.99,
  "activo": true
}

// Listar productos activos
GET http://localhost:3000/products?activos=true

// Soft delete de producto
DELETE http://localhost:3000/products/1

// Upsert masivo desde API externa
POST http://localhost:3000/products/sync
Content-Type: application/json

[
  { "id": 10, "descripcion": "Jean slim", "precio": 25999.90, "activo": true },
  { "descripcion": "Campera liviana", "precio": 45999.00 }
]

// Crear ticket
POST http://localhost:3000/tickets
Content-Type: application/json

{
  "nombreCliente": "Juan Perez",
  "items": [
    { "productId": 10, "cantidad": 1 },
    { "productId": 2, "cantidad": 3 }
  ]
}

// Consultar tickets
GET http://localhost:3000/tickets

// Consultar ticket por id
GET http://localhost:3000/tickets/1
```
